    <script>
        // ضع رابط Web App الخاص بك هنا
        const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbwmaMkSonVcBbzUX5Zb7ADrPS9bohhOSFpCvOVR84y-tiLARedNTIO0-i0nEFDGutedWw/exec"; 
        let fileBase64 = null;
        let fileNameVal = "";

        function toggleMedicalSection(show) {
            const medicalSection = document.getElementById('medicalSection');
            const canMeasureInputs = document.querySelectorAll('input[name="canMeasure"]');
            
            if (show) {
                medicalSection.classList.remove('hidden');
                canMeasureInputs.forEach(i => i.required = true);
            } else {
                medicalSection.classList.add('hidden');
                canMeasureInputs.forEach(i => {
                    i.required = false;
                    i.checked = false;
                });
            }
        }

        function validateNationalId(input) {
            const idError = document.getElementById('idError');
            input.value = input.value.replace(/[^0-9]/g, '');
            
            if (input.value.length > 0 && input.value.length !== 14) {
                idError.classList.remove('hidden');
                input.setCustomValidity('الرقم القومي يجب أن يتكون من 14 رقم بالضبط');
            } else {
                idError.classList.add('hidden');
                input.setCustomValidity('');
            }
        }

        function handleFile(input) {
            const fileNameElem = document.getElementById('fileName');
            if (input.files && input.files[0]) {
                const file = input.files[0];
                if (file.type !== 'application/pdf') {
                    alert('عذراً، يرجى اختيار ملف بصيغة PDF فقط.');
                    input.value = '';
                    fileNameElem.innerText = 'الحد الأقصى 10 ميجابايت';
                    return;
                }
                
                fileNameVal = file.name;
                const reader = new FileReader();
                reader.onload = function(e) {
                    fileBase64 = e.target.result;
                };
                reader.readAsDataURL(file);

                fileNameElem.innerText = `تم اختيار: ${file.name}`;
                fileNameElem.classList.add('text-sky-600', 'font-bold');
            }
        }

        async function handleSubmit(e) {
            e.preventDefault();

            const idInput = document.getElementById('nationalId');
            if (idInput.value.length !== 14) {
                document.getElementById('idError').classList.remove('hidden');
                idInput.focus();
                return;
            }

            const selectedRole = document.querySelector('input[name="role"]:checked')?.value;
            if (!selectedRole) {
                alert('برجاء اختيار مجال مشاركة واحد.');
                return;
            }

            const submitBtn = document.getElementById('submitBtn');
            const originalBtnHtml = submitBtn.innerHTML;
            submitBtn.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i> <span>جاري الإرسال...</span>`;
            submitBtn.disabled = true;

            const formData = {
                fullName: document.getElementById('fullName').value,
                phoneNumber: document.getElementById('phoneNumber').value,
                nationalId: idInput.value,
                isMedical: document.querySelector('input[name="isMedical"]:checked')?.value === 'yes' ? 'نعم' : 'لا',
                canMeasure: document.querySelector('input[name="canMeasure"]:checked')?.value === 'yes' ? 'نعم' : 'لا',
                role: selectedRole,
                fileData: fileBase64,
                fileName: fileNameVal
            };

            try {
                await fetch(SCRIPT_URL, {
                    method: 'POST',
                    mode: 'no-cors',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(formData)
                });

                document.getElementById('successModal').classList.remove('hidden');
                document.getElementById('volunteerForm').reset();
                document.getElementById('medicalSection').classList.add('hidden');
                document.getElementById('fileName').innerText = 'الحد الأقصى 10 ميجابايت';
                document.getElementById('fileName').classList.remove('text-sky-600', 'font-bold');
                fileBase64 = null;

            } catch (error) {
                alert('حدث خطأ أثناء الإرسال، برجاء المحاولة لاحقاً.');
                console.error(error);
            } finally {
                submitBtn.innerHTML = originalBtnHtml;
                submitBtn.disabled = false;
            }
        }
    </script>
